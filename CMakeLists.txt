cmake_minimum_required(VERSION 3.20)
project(AegisCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aegis_core INTERFACE)
target_include_directories(aegis_core INTERFACE core/include)
add_executable(aegis_tests core/tests/tests.cpp)
target_link_libraries(aegis_tests PRIVATE aegis_core)
add_executable(aegis_sim core/src/main.cpp)
target_link_libraries(aegis_sim PRIVATE aegis_core)
if(MSVC)
 target_compile_options(aegis_tests PRIVATE /W4 /WX)
 target_compile_options(aegis_sim PRIVATE /W4 /WX)
else()
 target_compile_options(aegis_tests PRIVATE -Wall -Wextra -Werror -pedantic)
 target_compile_options(aegis_sim PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME aegis_core COMMAND aegis_tests)
